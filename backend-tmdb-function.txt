const CORS = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, apikey, content-type, x-client-info',
  'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
};
const J = (b: unknown, s = 200) =>
  new Response(JSON.stringify(b), { status: s, headers: { ...CORS, 'Content-Type': 'application/json' } });
const G: Record<number, string> = {28:'اکشن',12:'ماجراجویی',16:'انیمیشن',35:'کمدی',80:'جنایی',99:'مستند',18:'درام',10751:'خانوادگی',14:'فانتزی',36:'تاریخی',27:'ترسناک',10402:'موسیقی',9648:'معمایی',10749:'عاشقانه',878:'علمی‌تخیلی',53:'هیجانی',10752:'جنگی',37:'وسترن',10759:'اکشن و ماجراجویی',10762:'کودکان',10764:'واقعیت',10765:'علمی‌تخیلی و فانتزی',10768:'جنگ و سیاست'};

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: CORS });

  // پروکسی تصویر: برای کاربرانی که image.tmdb.org برایشان باز نمی‌شود
  const url = new URL(req.url);
  const ip = url.searchParams.get('img');
  if (req.method === 'GET' && ip) {
    const w = url.searchParams.get('w') || 'w185';
    if (!/^\/[A-Za-z0-9_-]+\.(jpg|png)$/.test(ip) || !['w92', 'w154', 'w185', 'w342', 'w500', 'w780'].includes(w)) {
      return new Response('bad request', { status: 400, headers: CORS });
    }
    const r = await fetch('https://image.tmdb.org/t/p/' + w + ip);
    if (!r.ok) return new Response('not found', { status: 404, headers: CORS });
    return new Response(r.body, {
      headers: { ...CORS, 'Content-Type': r.headers.get('Content-Type') || 'image/jpeg', 'Cache-Control': 'public, max-age=604800, immutable' },
    });
  }

  const auth = req.headers.get('Authorization') || '';
  const apikey = req.headers.get('apikey') || '';
  const u = await fetch(Deno.env.get('SUPABASE_URL') + '/auth/v1/user', { headers: { Authorization: auth, apikey } });
  if (!u.ok) return J({ error: 'unauthorized' }, 401);
  const key = Deno.env.get('TMDB_API_KEY');
  if (!key) return J({ error: 'no key' }, 500);
  let b: any = {};
  try { b = await req.json(); } catch { /* empty */ }
  const bearer = key.length > 60;
  const call = async (path: string, params: Record<string, string> = {}) => {
    const q = new URLSearchParams({ language: 'fa-IR', ...params });
    if (!bearer) q.set('api_key', key);
    const r = await fetch('https://api.themoviedb.org/3' + path + '?' + q, { headers: bearer ? { Authorization: 'Bearer ' + key } : {} });
    if (!r.ok) throw new Error('tmdb ' + r.status);
    return r.json();
  };
  // نتیجهٔ فارسی + نام انگلیسی + خلاصهٔ انگلیسی (اگر فارسی نبود)
  const both = async (path: string, params: Record<string, string> = {}) => {
    const [f, e] = await Promise.all([call(path, params), call(path, { ...params, language: 'en-US' })]);
    const k = (x: any) => (x.media_type || '') + x.id;
    const m = new Map(e.results.map((x: any) => [k(x), x]));
    f.results.forEach((x: any) => {
      const y: any = m.get(k(x));
      x.en = y ? (y.title || y.name) : (x.original_title || x.original_name || '');
      if (!x.overview && y) x.overview = y.overview || '';
    });
    return f;
  };
  const map = (x: any, t?: string) => {
    const mt = x.media_type || t;
    const nm = x.title || x.name;
    return {
      id: 't' + mt + x.id, t: nm, en: x.en && x.en !== nm ? x.en : '', k: mt === 'tv' ? 'series' : 'film',
      o: x.original_language === 'fa' ? 'ایرانی' : 'خارجی',
      y: (x.release_date || x.first_air_date || '').slice(0, 4),
      g: (x.genre_ids || []).map((i: number) => G[i]).filter(Boolean).slice(0, 2).join('، '),
      d: x.overview || '',
      img: x.poster_path ? 'https://image.tmdb.org/t/p/w185' + x.poster_path : '',
      bd: x.backdrop_path ? 'https://image.tmdb.org/t/p/w780' + x.backdrop_path : '',
      r: x.vote_average ? Math.round(x.vote_average * 10) / 10 : 0,
    };
  };
  try {
    if (b.action === 'search') {
      const q = String(b.q || '').trim();
      if (q.length < 2) return J({ results: [] });
      const d = await both('/search/multi', { query: q, include_adult: 'false' });
      return J({ results: d.results.filter((x: any) => x.media_type === 'movie' || x.media_type === 'tv').slice(0, 10).map((x: any) => map(x)) });
    }
    if (b.action === 'home') {
      const [tr, np, up, tv] = await Promise.all([both('/trending/all/day'), both('/movie/now_playing'), both('/movie/upcoming'), both('/tv/popular')]);
      const f = (d: any, t?: string) => d.results.filter((x: any) => x.poster_path || x.backdrop_path).slice(0, 12).map((x: any) => map(x, t));
      return J({ trending: f(tr), now: f(np, 'movie'), upcoming: f(up, 'movie'), tv: f(tv, 'tv') });
    }
    if (b.action === 'smart') {
      const q = String(b.q || '').trim();
      if (q.length < 2) return J({ results: [], people: [], both: null });
      const isMedia = (x: any) => x.media_type === 'movie' || x.media_type === 'tv';
      // ۱) عنوان‌های مشابه؛ اگر نتیجه کم بود، تک‌تک کلمات هم جستجو می‌شوند
      const d = await both('/search/multi', { query: q, include_adult: 'false' });
      let titles: any[] = d.results.filter(isMedia);
      const words = q.split(/\s+/).filter((w) => w.length >= 3).slice(0, 4);
      if (titles.length < 5 && words.length >= 2) {
        const extra = await Promise.all(words.map((w) => both('/search/multi', { query: w, include_adult: 'false' }).catch(() => ({ results: [] }))));
        const seen = new Map<string, any>();
        for (const x of [...titles, ...extra.flatMap((e: any) => e.results.filter(isMedia))]) seen.set(x.media_type + x.id, x);
        const lw = words.map((w) => w.toLowerCase());
        const score = (x: any) => lw.filter((w) => ((x.title || x.name || '') + ' ' + (x.en || '')).toLowerCase().includes(w)).length;
        titles = [...seen.values()].sort((a: any, c: any) => score(c) - score(a) || (c.popularity || 0) - (a.popularity || 0));
      }
      // ۲) بازیگر یا کارگردان (چند اسم با , یا + یا «و» جدا می‌شوند)
      const segs = q.split(/\s*(?:[,+،&؛;]|\s+و\s+|\s+and\s+)\s*/i).map((s) => s.trim()).filter((s) => s.length >= 2).slice(0, 3);
      const found = await Promise.all((segs.length ? segs : [q]).map(async (sg) => {
        try {
          const pr = await call('/search/person', { query: sg });
          const p = (pr.results || [])[0];
          return p && (p.popularity || 0) >= 2 ? p : null;
        } catch { return null; }
      }));
      const persons: any[] = [];
      for (const p of found) if (p && !persons.some((x) => x.id === p.id)) persons.push(p);
      const cred = async (p: any) => {
        const [f, e] = await Promise.all([
          call('/person/' + p.id + '/combined_credits'),
          call('/person/' + p.id + '/combined_credits', { language: 'en-US' }),
        ]);
        const enm = new Map<string, string>();
        [...(e.cast || []), ...(e.crew || [])].forEach((x: any) => enm.set(x.media_type + x.id, x.title || x.name));
        const seen = new Map<string, any>();
        [...(f.cast || []), ...(f.crew || []).filter((x: any) => x.job === 'Director')].forEach((x: any) => {
          if (x.media_type !== 'movie' && x.media_type !== 'tv') return;
          if ((x.genre_ids || []).some((g: number) => g === 10767 || g === 10763 || g === 10764)) return;
          if (/Self|Himself|Herself/i.test(x.character || '')) return;
          const k = x.media_type + x.id;
          if (!seen.has(k)) { x.en = enm.get(k) || x.original_title || x.original_name || ''; seen.set(k, x); }
        });
        return { p, list: [...seen.values()] };
      };
      const cs = await Promise.all(persons.map(cred));
      const byPop = (a: any[]) => a.sort((x: any, y: any) => (y.popularity || 0) - (x.popularity || 0));
      let bothRes: any = null;
      if (cs.length >= 2) {
        const sets = cs.map((c) => new Set(c.list.map((x: any) => x.media_type + x.id)));
        const common = cs[0].list.filter((x: any) => sets.every((s) => s.has(x.media_type + x.id)));
        bothRes = { names: cs.map((c) => c.p.name), items: byPop(common).slice(0, 20).map((x: any) => map(x)) };
      }
      const people = cs.map((c) => ({ name: c.p.name, items: byPop(c.list).slice(0, cs.length >= 2 ? 6 : 15).map((x: any) => map(x)) }));
      return J({ results: titles.slice(0, 10).map((x: any) => map(x)), people, both: bothRes });
    }
    if (b.action === 'providers') {
      const mt = b.mt === 'tv' ? 'tv' : 'movie';
      const id = String(b.id || '').replace(/\D/g, '');
      if (!id) return J({ error: 'bad id' }, 400);
      const d = await call('/' + mt + '/' + id + '/watch/providers');
      const out: Record<string, any> = {};
      for (const [cc, v] of Object.entries(d.results || {}) as [string, any][]) {
        const pick = (a?: any[]) => (a || []).slice(0, 12).map((p: any) => ({ n: p.provider_name, l: p.logo_path || '' }));
        out[cc] = { link: v.link || '', flatrate: pick(v.flatrate), free: pick(v.free), ads: pick(v.ads), rent: pick(v.rent), buy: pick(v.buy) };
      }
      return J({ results: out });
    }
    return J({ error: 'bad action' }, 400);
  } catch (e) {
    return J({ error: String(e) }, 502);
  }
});
